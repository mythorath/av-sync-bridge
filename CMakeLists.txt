if(AVSYNC_BUILD_NETWORK)
  find_package(Threads REQUIRED)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GSTREAMER REQUIRED IMPORTED_TARGET
    gstreamer-1.0>=1.24 gstreamer-net-1.0>=1.24 gstreamer-app-1.0>=1.24
    gstreamer-audio-1.0>=1.24 gstreamer-rtp-1.0>=1.24 gio-2.0)
  if(WIN32)
    get_filename_component(_gst_runtime_default "${GSTREAMER_gstreamer-1.0_PREFIX}/bin" ABSOLUTE)
    set(AVSYNC_GSTREAMER_RUNTIME_DIR "${_gst_runtime_default}" CACHE PATH
      "Matching GStreamer DLL directory prepended only to CTest child processes")
    foreach(_gst_dll gstreamer-1.0-0.dll gstaudio-1.0-0.dll)
      if(NOT EXISTS "${AVSYNC_GSTREAMER_RUNTIME_DIR}/${_gst_dll}")
        message(FATAL_ERROR "Missing ${_gst_dll}: set AVSYNC_GSTREAMER_RUNTIME_DIR to the matching SDK bin directory")
      endif()
    endforeach()
  endif()
  add_library(avsync_gst_clock src/net/network_clock.cpp)
  target_include_directories(avsync_gst_clock PUBLIC include)
  target_link_libraries(avsync_gst_clock PUBLIC avsync_core PkgConfig::GSTREAMER)
  add_library(avsync_rtp_anchor src/net/rtp_audio_anchor.cpp)
  target_include_directories(avsync_rtp_anchor PUBLIC include)
  target_link_libraries(avsync_rtp_anchor PUBLIC avsync_core PkgConfig::GSTREAMER)
  add_library(avsync_nominal_audio src/audio/nominal_audio.cpp)
  target_include_directories(avsync_nominal_audio PUBLIC include)
  target_link_libraries(avsync_nominal_audio PUBLIC PkgConfig::GSTREAMER)
  add_executable(avsync-nominal-audio-probe apps/nominal_audio_probe.cpp)
  target_link_libraries(avsync-nominal-audio-probe PRIVATE avsync_nominal_audio avsync_core)
  if(MSVC)
    target_compile_options(avsync_nominal_audio PRIVATE /W4 /permissive-)
  else()
    target_compile_options(avsync_nominal_audio PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(avsync-nominal-audio-probe PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_executable(avsync-conversion-timing-probe apps/conversion_timing_probe.cpp)
  target_link_libraries(avsync-conversion-timing-probe PRIVATE avsync_core PkgConfig::GSTREAMER Threads::Threads)
  if(NOT MSVC)
    target_compile_options(avsync-conversion-timing-probe PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  if(BUILD_TESTING)
    add_test(NAME avsync-conversion-timing-help COMMAND avsync-conversion-timing-probe --help)
  endif()
  if(MSVC)
    target_compile_options(avsync_gst_clock PRIVATE /W4 /permissive-)
  else()
    target_compile_options(avsync_gst_clock PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  if(WIN32)
    add_executable(avsync-windows-sender apps/windows_sender.cpp)

# CTest can run from a fresh shell, not just the SDK-equipped build shell.
# Apply within each declaring directory (compatible with CMake 3.24). Python
# fixture children inherit this test-local PATH; no user/system PATH is edited.
function(avsync_configure_test_runtime)
  if(WIN32 AND AVSYNC_BUILD_NETWORK)
    get_property(_avsync_tests DIRECTORY PROPERTY TESTS)
    foreach(_avsync_test IN LISTS _avsync_tests)
      set_property(TEST "${_avsync_test}" APPEND PROPERTY ENVIRONMENT_MODIFICATION
        "PATH=path_list_prepend:${AVSYNC_GSTREAMER_RUNTIME_DIR}")
    endforeach()
  endif()
endfunction()
if(BUILD_TESTING)
  add_subdirectory(tests)
  avsync_configure_test_runtime()
endif()
