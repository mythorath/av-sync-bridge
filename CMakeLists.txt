cmake_minimum_required(VERSION 3.24)
project(av_sync_bridge VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(CTest)
set(AVSYNC_IPC_DEFAULT OFF)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(AVSYNC_IPC_DEFAULT ON)
endif()
option(AVSYNC_BUILD_IPC "Build Linux IPC and synthetic tools" ${AVSYNC_IPC_DEFAULT})
option(AVSYNC_BUILD_OBS_PLUGIN "Build experimental Linux OBS adapter" OFF)
option(AVSYNC_BUILD_OBS_SMOKE "Build isolated libOBS synthetic recording harness" OFF)
option(AVSYNC_ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(AVSYNC_BUILD_WINDOWS_PROBE "Build metadata-only WASAPI loopback probe" ${WIN32})

if(AVSYNC_ENABLE_SANITIZERS)
  if(MSVC)
    message(FATAL_ERROR "Use a GCC/Clang Linux build for the sanitizer option")
  endif()
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()

# Source lists are explicit so adding a local file cannot silently publish/build it.
add_library(avsync_core src/core/timing.cpp)
target_include_directories(avsync_core PUBLIC include)
if(MSVC)
  target_compile_options(avsync_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(avsync_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(AVSYNC_BUILD_WINDOWS_PROBE)
  if(NOT WIN32)
    message(FATAL_ERROR "The WASAPI probe requires Windows")
  endif()
  add_executable(avsync-wasapi-probe apps/wasapi_probe.cpp)
  target_compile_definitions(avsync-wasapi-probe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(avsync-wasapi-probe PRIVATE ole32 uuid)
  if(MSVC)
    target_compile_options(avsync-wasapi-probe PRIVATE /W4 /permissive-)
  endif()
  if(BUILD_TESTING)
    # Help must not initialize COM, open an endpoint, or capture any audio.
    add_test(NAME avsync-wasapi-probe-help COMMAND avsync-wasapi-probe --help)
  endif()
endif()

if(AVSYNC_BUILD_IPC)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "IPC prototype currently supports Linux only")
  endif()
  find_package(Threads REQUIRED)
  add_library(avsync_ipc src/ipc/ipc.cpp)
  target_include_directories(avsync_ipc PUBLIC include)
  target_link_libraries(avsync_ipc PUBLIC Threads::Threads)
  target_compile_options(avsync_ipc PRIVATE -Wall -Wextra -Wpedantic)
  if(BUILD_TESTING)
    target_compile_definitions(avsync_ipc PRIVATE AVSYNC_IPC_TEST_HOOKS)
  endif()
  add_executable(avsync-synthetic apps/synthetic_daemon.cpp)
  target_link_libraries(avsync-synthetic PRIVATE avsync_ipc)
  add_executable(avsync-probe apps/probe.cpp)
  target_link_libraries(avsync-probe PRIVATE avsync_ipc)
endif()

if(AVSYNC_BUILD_OBS_PLUGIN)
  if(NOT AVSYNC_BUILD_IPC)
    message(FATAL_ERROR "OBS adapter requires AVSYNC_BUILD_IPC")
  endif()
  add_subdirectory(plugins/obs)
endif()

if(AVSYNC_BUILD_OBS_SMOKE)
  if(NOT AVSYNC_BUILD_OBS_PLUGIN)
    message(FATAL_ERROR "OBS smoke harness requires AVSYNC_BUILD_OBS_PLUGIN")
  endif()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SMOKE_OBS REQUIRED IMPORTED_TARGET libobs>=32.2)
  pkg_check_modules(X11 REQUIRED IMPORTED_TARGET x11)
  get_target_property(_smoke_options PkgConfig::SMOKE_OBS INTERFACE_COMPILE_OPTIONS)
  if(_smoke_options)
    list(FILTER _smoke_options EXCLUDE REGEX "^-std=")
    set_property(TARGET PkgConfig::SMOKE_OBS PROPERTY INTERFACE_COMPILE_OPTIONS "${_smoke_options}")
  endif()
  add_executable(avsync-obs-smoke tools/obs_smoke.cpp)
  target_link_libraries(avsync-obs-smoke PRIVATE PkgConfig::SMOKE_OBS PkgConfig::X11 Threads::Threads)
  find_program(AVSYNC_OBS_MUX_HELPER NAMES obs-ffmpeg-mux REQUIRED)
  add_custom_command(TARGET avsync-obs-smoke POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E create_symlink "${AVSYNC_OBS_MUX_HELPER}"
      "$<TARGET_FILE_DIR:avsync-obs-smoke>/obs-ffmpeg-mux"
    COMMENT "Link the installed OBS mux helper beside the isolated test executable")
endif()

if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
